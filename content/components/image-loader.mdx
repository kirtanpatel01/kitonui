## Usage

```tsx
import { ImageGenerationLoader } from "@/components/ui/image-loader"

export default function Demo() {
  return (
    <div className="relative overflow-hidden rounded-lg">
      <img
        src="https://images.unsplash.com/photo-1620694563886-c3a80ec55f41"
        alt="Demo"
        className="w-full object-cover"
      />
      <ImageGenerationLoader text="GENERATING" effect="scale-wave" />
    </div>
  )
}
```

## Props

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `text` | `string` | `"GENERATING"` | Overlay label text rendered above the pixel grid |
| `effect` | `"scale-wave" \| "fade" \| "slide"` | `"scale-wave"` | Animation effect for the pixel dissolve |
| `gridSize` | `number` | `24` | Number of pixel columns in the canvas grid |
| `duration` | `number` | `2000` | Total animation duration in milliseconds |
| `className` | `string` | — | Additional class names for the outer container |

<Callout type="info" title="CORS Requirement">
  The image must allow cross-origin resource sharing (CORS). If you pass an image that blocks CORS, the canvas color extraction will silently fall back to the default palette.
</Callout>
