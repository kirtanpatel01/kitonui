## Usage

```tsx
import { ShimmerText } from "@/components/ui/shimmer-text"

export default function Demo() {
  return (
    <ShimmerText duration={4}>
      KitonUi
    </ShimmerText>
  )
}
```

## Props

| Prop | Type | Default | Description |
| --- | --- | --- | --- |
| `text` | `string` | — | Text to display (can alternatively pass `children`) |
| `duration` | `number` | `4` | Full cycle animation duration in seconds |
| `className` | `string` | — | Additional CSS classes for typography, sizing, and colors |
